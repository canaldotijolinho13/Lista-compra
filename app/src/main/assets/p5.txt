<script>
(function(){
  var KEY='lista_tema';
  function aplicar(tema){
    document.body.classList.toggle('dark', tema==='escuro');
    var b=document.getElementById('tema');
    if(b) b.textContent = tema==='escuro' ? 'Modo claro' : 'Modo escuro';
  }
  var h1=document.querySelector('h1');
  var sub=document.getElementById('sub');
  var head=document.createElement('div');
  head.className='head';
  var left=document.createElement('div');
  left.className='left';
  h1.parentNode.insertBefore(head,h1);
  left.appendChild(h1);
  left.appendChild(sub);
  var btn=document.createElement('button');
  btn.id='tema';
  btn.type='button';
  btn.textContent='Modo escuro';
  head.appendChild(left);
  head.appendChild(btn);
  aplicar(localStorage.getItem(KEY)==='escuro' ? 'escuro' : 'claro');
  btn.addEventListener('click', function(){
    var novo = document.body.classList.contains('dark') ? 'claro' : 'escuro';
    localStorage.setItem(KEY, novo);
    aplicar(novo);
  });
})();
</script>
